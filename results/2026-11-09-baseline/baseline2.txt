#!/bin/sh

case "$1" in
        start)
                printf "Turning on WiFI Power: "
                /bin/echo 14 > /sys/class/gpio/export
                /bin/echo out > /sys/class/gpio/gpio14/direction
                [ $? = 0 ] && echo "OK" || echo "FAIL"
                ;;
        stop)
                printf "Turning off WiFI Power: "
                /bin/echo 14 > /sys/class/gpio/unexport
                [ $? = 0 ] && echo "OK" || echo "FAIL"
                ;;
        restart|reload)
                $0 stop
                $0 start
                ;;
        *)
                echo "Usage: $0 {start|stop|restart}"
                exit 1
esac

#!/bin/sh

        start)
                printf "Starting wpa_supplicant: "
                /usr/sbin/wpa_supplicant -B -Dnl80211 -iwlan0 -c /etc/wpa_supplicant.conf  &> /tmp/wpa.info &

                [ $? = 0 ] && echo "OK" || echo "FAIL"

                [ $? = 0 ] && echo "OK" || echo "FAIL"
                ;;

                ;;
